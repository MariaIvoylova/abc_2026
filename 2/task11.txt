#  todo: Дан номер месяца (1 — январь, 2 — февраль, ...). Вывести название соответствующего
#  времени года ("зима", "весна" и т.д.).
month = int(input("1–12"))

if month in (12, 1, 2):
    season = "зима"
elif month in (3, 4, 5):
    season = "весна"
elif month in (6, 7, 8):
    season = "лето"
elif month in (9, 10, 11):
    season = "осень"
else:
    season = "неверный номер месяца"

print(season)
